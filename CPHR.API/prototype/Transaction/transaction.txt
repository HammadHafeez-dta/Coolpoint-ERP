        public static bool UpdateLeaveAdjustment(Leave entity)
        {
            try
            {
                var db = DB.CreateMainHRMS();
                {
                    var temp = db.HRM_Leave_Register.Where(u => u.Emp_CardNo == entity.EmpCardNo && u.Leave_Month == entity.LeaveDate.Month && u.Leave_Year == entity.LeaveDate.Year && u.Application_No == 0 && u.Site == Constants.SITE).FirstOrDefault();
                    if (temp == null)
                    {
                        temp = new HRM_Leave_Register();
                        temp.Emp_CardNo = entity.EmpCardNo;
                        temp.Leave_date = entity.LeaveDate;
                        temp.Leave_Month = entity.LeaveDate.Month;
                        temp.Leave_Year = entity.LeaveDate.Year;
                        temp.Application_No = 0;

                        temp.AL_Open = entity.ALOpen;
                        temp.CL_Open = entity.CLOpen;
                        temp.SL_Open = entity.SLOpen;
                        temp.CPL_Open = entity.CPLOpen;

                        temp.Site = Constants.SITE;
                        temp.Enter_By = Constants.USER_ID;
                        temp.Entery_Time = DateTime.Now;
                        db.HRM_Leave_Register.Add(temp);
                        db.SaveChanges();
                        return true;
                    }
                    else
                    {
                        temp.Leave_date = entity.LeaveDate;

                        temp.AL_Open = entity.ALOpen;
                        temp.CL_Open = entity.CLOpen;
                        temp.SL_Open = entity.SLOpen;
                        temp.CPL_Open = entity.CPLOpen;

                        temp.Edit_By = Constants.USER_ID;
                        temp.Edit_Time = DateTime.Now;
                        db.SaveChanges();
                        return true;
                    }
                }
            }
            catch (Exception)
            {
                return false;
            }
        }

        public static int SaveLoan(Loan entity)

                        temp = new HRM_Loans();

                        temp.Loan_Type = entity.LoanType;
                        temp.Loan_Date = entity.LoanDate;
                        temp.Emp_CardNo = entity.EmpCardNo;
                        temp.Loan_Amount = entity.LoanAmount;
                        temp.Loan_Installment = entity.InstallmentAmount;
                        temp.Entery_Time = DateTime.Now;
                        temp.site = Constants.SITE;
                        temp.Enter_By = Constants.USER_ID;
                        db.HRM_Loans.Add(temp);
                        db.SaveChanges();
                    }
                    else
                    {
                        temp.Loan_Type = entity.LoanType;
                        temp.Loan_Date = entity.LoanDate;
                        temp.Emp_CardNo = entity.EmpCardNo;
                        temp.Loan_Amount = entity.LoanAmount;
                        temp.Loan_Installment = entity.InstallmentAmount;
                        temp.Edit_Time = DateTime.Now;
