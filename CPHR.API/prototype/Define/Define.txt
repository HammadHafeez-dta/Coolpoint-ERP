        public static void UpdateEmployeeSalary(Employee entity)
        {
            try
            {

        public static List<HRM_vew_Employee> GetEmployeeList(string type, int id, string name)
        {
            try
            {
                var db = DB.CreateMainHRMS();
                if (type == "Employee Name Wise")
                    return (db.HRM_vew_Employee.Where(u => u.Status == true && u.site == Constants.SITE && u.Emp_Name.Contains(name)).ToList());
                else if (type == "Department Wise")
                    return (db.HRM_vew_Employee.Where(u => u.Dept_Code == id && u.Status == true && u.site == Constants.SITE).OrderBy(o => o.Emp_CardNo).ToList());
                else if (type == "Designation Wise")
                    return (db.HRM_vew_Employee.Where(u => u.Status == true && u.site == Constants.SITE && u.Designation_Name.Contains(name)).ToList());
                else if (type == "Scale Wise")
                    return (db.HRM_vew_Employee.Where(u => u.Status == true && u.site == Constants.SITE && u.Scale ==name).ToList());
                else if (type == "Employee Card Wise")
                    return (db.HRM_vew_Employee.Where(u => u.Emp_CardNo == id && u.Status == true && u.site == Constants.SITE).OrderBy(o => o.Emp_CardNo).ToList());
                else
                    return (db.HRM_vew_Employee.Where(u => u.Status == true && u.site == Constants.SITE).OrderBy(o => o.Emp_CardNo).ToList());
            }
            catch (Exception ex)
            {
                return null;
            }
        }
