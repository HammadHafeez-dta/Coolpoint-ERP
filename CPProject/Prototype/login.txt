        public static String EncryptPassword(String text)
        {
            String encText = String.Empty;
            int ascode = 0;
            for (int i = 0; i < text.Length; i++)
            {
                ascode = Convert.ToInt32(text[i]); //'Get the Asc Code of the Character stored in Reply
                // encText += Convert.ToString(((ascode * 20)/2)-100);

                if (i == 0)
                    encText = Convert.ToString(Convert.ToInt32((Math.Ceiling(Convert.ToDouble((ascode * 29) / 3))) - 102), 16);
                else
                {
                    //encText += "-" + Convert.ToString((((ascode * 29) / 3) - 102), 16);
                    encText += "-" + Convert.ToString(Convert.ToInt32((Math.Ceiling(Convert.ToDouble((ascode * 29) / 3))) - 102), 16);
                }
            }
            return encText;
        }
        public static String DeEncryptPassword(String text)
        {
            String decText = String.Empty;
            String[] chunks = text.Split('-');

            if (chunks != null && chunks.Length > 1)
            {
                foreach (String chunk in chunks)
                {

                    decText += Convert.ToChar(Convert.ToInt32(Math.Ceiling(((Convert.ToInt32(chunk, 16) + 102) * 3) / 29.0)));
                }
            }
            return decText;
        }
